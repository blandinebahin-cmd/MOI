//////////////////////////////////////////////////////////////////////////////// Prime sur Investissement ///////////////////////////////////////////////////////////////////////////////////////

PInvestissement = SaisieM1("P.Invest.fixe")
TopRattachementAnnuelPInvestissement = Saisie("Top.Rattache.Invest",0)

If PInvestissement <> 0 then

	///// Spécifique - Rattachement de la prime sur dates execices comptables pour la prime objectif versée en décembre si EV TopRattachementAnnuel = 1 sinon rattachement au mois en cours

	If TopRattachementAnnuelPInvestissement = 1 then

		///// Rattachement de la prime sur dates execices comptables
		DateDebAn = date(1,10,bul.annee-1)
		DateFinAn = date(30,09,bul.annee)

		If MtPart.EXERCICEFISCALSPE = 1 then
			DateDebAn = date(1,07,bul.annee-1)
			DateFinAn = date(30,06,bul.annee)
		Endif


		//prise en compte des entrées en cours d'année
		If emp.dtdeb > DateDebAn Then
			DateDebAn = emp.dtdeb
		Endif

		//prise en compte des entrées en cours d'année
		If bul.sorticemois = true then
			DateFinAn = emp.dtfin
		Endif
	Endif


	///// Execution de la prime

	Exec("Lprime_H04.ATE")
	methodeCalcul = 2
	Bases = PInvestissement
	Liblong = "Prime sur investissement"

	If TopRattachementAnnuelPInvestissement = 1 then
		marquedtdeb = DateDebAn
		marquedtfin = DateFinAn

		//print "PPrime P.FIXE- Date Début AED Prime investissement fixe : " + DateDebAn
		//print "PPrime P.FIXE- Date Fin AED Prime investissement fixe : " + DateFinAn
	Endif

	Exec("GenereLprime")
EndIf
